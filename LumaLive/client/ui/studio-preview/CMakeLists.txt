cmake_minimum_required(VERSION 3.24)
# Build the real preview client independently of the full server/AI dependency
# graph. This also works as a subdirectory of the original LumaLive project.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(LumaLivePreview LANGUAGES CXX)
  enable_testing()
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
  add_compile_definitions($<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=0> $<$<CONFIG:Debug>:NDEBUG>)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_subdirectory(../../media-pipeline media-pipeline)
  add_subdirectory(../../../shared shared)
  get_filename_component(LUMALIVE_WEBRTC_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../third_party/webrtc" ABSOLUTE)
  add_subdirectory(../../webrtc webrtc)
  add_subdirectory(../../signaling signaling)
  add_subdirectory(../../../server/signaling signaling-server)
  add_executable(luma_signaling_server ../../../apps/luma-server/src/main.cpp)
  target_link_libraries(luma_signaling_server PRIVATE luma_signaling)
  set(LUMA_PREVIEW_STANDALONE ON)
endif()
if(NOT WIN32)
  message(FATAL_ERROR "The studio preview uses Windows Media Foundation.")
endif()
add_library(luma_studio_preview STATIC src/StudioPreview.cpp)
target_include_directories(luma_studio_preview PUBLIC include)
target_compile_features(luma_studio_preview PUBLIC cxx_std_20)
target_compile_options(luma_studio_preview PRIVATE /utf-8 /W4 /permissive-)
target_link_libraries(luma_studio_preview PUBLIC luma_client_media_pipeline PRIVATE user32 gdi32 comctl32 comdlg32 mfplay mfplat mfuuid ole32 winmm dwmapi)
target_link_libraries(luma_studio_preview PUBLIC luma_client_webrtc luma_client_signaling)
if(LUMA_PREVIEW_STANDALONE)
  add_executable(luma_studio WIN32 src/main.cpp)
  target_link_libraries(luma_studio PRIVATE luma_studio_preview)
endif()
add_executable(luma_preview_media_tests tests/PreviewMediaTests.cpp)
target_include_directories(luma_preview_media_tests PRIVATE include)
target_link_libraries(luma_preview_media_tests PRIVATE luma_client_media_pipeline)
add_test(NAME luma_preview_media_tests COMMAND luma_preview_media_tests)
add_executable(luma_preview_call_tests tests/PreviewCallTests.cpp)
target_include_directories(luma_preview_call_tests PRIVATE include)
target_link_libraries(luma_preview_call_tests PRIVATE luma_client_webrtc luma_client_signaling luma_client_media_pipeline luma_signaling ole32)
add_test(NAME luma_preview_call_tests COMMAND luma_preview_call_tests)
set_tests_properties(luma_preview_call_tests PROPERTIES TIMEOUT 35)

add_executable(luma_file_playback_tests tests/FilePlaybackTests.cpp)
target_compile_options(luma_file_playback_tests PRIVATE /utf-8)
target_link_libraries(luma_file_playback_tests PRIVATE mfplay mfplat mfuuid ole32 user32)
